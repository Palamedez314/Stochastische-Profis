\subsection*{Aufgabe 10.}

Sei \WraumAnn{}, $\F=(\F_n)_{n\ge 0}$ eine Filtration und $X=(X_n)_{n\ge 0}$ ein an \F{} adaptierter Stochastischer Prozess mit $X_n\in\scriptL_1(P)$ für alle $n\ge 0$.
\bigskip

Zeigen Sie: Wenn $H.X$ ein $\F$-Martingal für alle $\F$-vorhersagbaren Prozesse $H=(H_n)_{n\ge 1}$ mit $H_n\in\scriptL_{\infty}(P)$ für alle $n\ge 1$, dann ist $X$ auch ein $\F$-Martingal.
\bigskip

Unter welchen Bedingungen ist $X$ ein $\F$-Sub-Martingal?

\subsubsection*{Beweis/Lösung}
\einruecken

Sei X ein $\F$-adaptierter Prozess mit der genannten Eigenschaft. Definiere $H_n:=\1_{\Om}$. Dann $H:=(H_n)_{n\ge 1}$ offensichtlich $\F$-vorhersagbarer Prozess und $H\in\scriptL_{\infty}(P)$.
Wegen $H.X$ Martingal liefert
\begin{align*}
0 &= \E((H.X)_n-(H.X)_{n-1}|\F_{n-1}) \\
&= \E(H_n\cdot(X_n-X_{n-1})|\F_{n-1}) \\
&= \E(X_n-X_{n-1}|\F_{n-1}),
\end{align*}
dass $X$ auch ein $\F$-Martingal ist.

\item Verwende die Markov-Ungleichung und die Abschätzung aus \ref{11.ii} für $t>0$ beliebig. Optimiere anschließend bezüglich $t$.
\end{enumerate}

\subsection*{Aufgabe 12}

Seien $(D_n)_{n\ge 1}$ i.i.d. mit $P(D_1=-1)=P(D_1=1)=1/2$. Wir interpretieren jedes $D_i$ als Ergebnis einer fairen Wette, bei der im Misserfolgsfall der Einsatz in der nächsten Runde verdoppelt wird. Im Erfolgsfall wird dagegen das Spiel beendet. Dies beschreibt die folgende Verdopplungsstrategie
$$H_n:=2^{n-1}\1_{\{D_1=\dots=D_{n-1}=-1\}}, \quad n\ge 1$$
für den Einsatz in der $n$-ten Runde. Für diese Spielstrategie ist der Gesamtgewinn nach $n$ Runden durch
$$G_n=\sum_{i=1}^n H_iD_i$$
gegeben.

\begin{enumerate}[$i)$]
\item Finde ein Martingal $X=(X_n)_{n\ge 0}$ für das gilt $G=H.X$.
\item Zeige $\E G_n=0$ für alle $n\ge 1$.
\item Zeige $\lim_{n\rightarrow\infty}G_n=1$ fast sicher.
\end{enumerate}

\subsubsection*{Lösung}
\einruecken

\begin{enumerate}[$i)$,wide]
\item Sei $\F:=\F^D$. Da $H_n=2^{n-1}\1_U$ mit $U=\bigcap_{k=1}^{n-1}D_k^{-1}(\{-1\})\in\F_{n-1}$, ist $H$ $\F$-vorhersagbar. Sei nun der stochastische Prozess $X=(X_n)_{n\ge 0}$ definiert durch $X_0:=0, X_n:=\sum_{k=1}^n D_k$. Dann ist $X_n$ $\F_n$-messbar als Summe $\F_n$-messbarer Abbildungen, also $X$ $\F$-adaptiert.
Weiter gilt 
\begin{align*}
	\E(X_n-X_{n-1}|\F_{n-1}) &= \E(D_n|\F_{n-1})\\
	&= \E(D_n) = 0
\end{align*}
da $D_n$ unabhängig von $\F_{n-1}=\sigma(D_1,\dots,D_{n-1})$.
Somit ist $X$ Martingal bezüglich $\F$ mit
$$G_n=\sum_{i=1}^n H_iD_i=\sum_{i=1}^n H_i\cdot(X_i-X_{i-1})=(H.X)_{n}.$$

\item Beweis per Induktion:\\
Es gilt $H_1=1$, also
$$\E(G_1)=\E(D_1)=0.$$
Sei nun $1\le n<1$ mit $\E(G_n)=0$. Dann gilt
$$\E(G_{n+1})=\E(\E(G_{n+1}|\F_{n}))=\E(G_n)=0,$$
wobei der vorletzte Schritt daraus folgt, dass $G$ nach $i)$ ein Martingal ist. 

\item Seien $V_n:=\{D_1=\dots=D_n=-1\}$ und $V:=\bigcap_{n\ge 1}V_n$.
Dann gilt
\begin{align*}
P(V_n) &= P(\bigcap_{k=1}^n D_k^{-1}(\{-1\})) \\
&= \prod_{k=1}^n P(D_k^{-1} (\{-1\})) \tag{$\ast$} \\
&= \prod_{k=1}^n \frac{1}{2} \\
&= 2^{-n},
\end{align*}
wobei ($\ast$) wegen $(D_n)_{n\ge 1}$ i.i.d.
Somit folgt aus $V_n\searrow V$ und der Stetigkeit von oben
$$P(V)=\lim_{n\rightarrow \infty} P(V_n)=0.$$
Seien nun
$$W_n:=\{D_n\notin\{-1,1\}\},\quad W:=\bigcup_{n\ge 1}W_n,\quad N:=V\cup W.$$
Dann $P(W_n)=0$ nach Aufgabenstellung, also P(N) Nullmenge wegen
$$P(N)\le P(V)+P(W)\le P(V)+\sum_{n\ge 1}P(W_n)=0.$$
Für $\om\in\Om\setminus N=\bigcup_{n\ge 1}\Om\setminus (V_n\cup W)$ existiert $n\ge 1$ minimal, sodass $\om\in\Om\setminus V_n$. Per Definition gilt also $D_k(\om)=-1$ für $k<n$ und $D_n(\om)=1$. Somit
$$H_n(\om)=
\begin{cases}
2^{k-1} & k>n \\
0 & k>n,
\end{cases}$$
also für $m\ge n$:
\begin{align*}
G_m(\om) &= \sum_{k=1}^m{H_k(\om)\cdot D_k(\om)} \\
&= \left(\sum_{k=1}^{n-1} -2^{k-1}\right)+2^{n-1} \\
&= -(2^{n-1}-1)+2^{n-1} = 1,
\end{align*}
und somit $\lim_{n\rightarrow\infty} G_n(\om)=1$, wie gewünscht.
\end{enumerate}
\ausruecken
